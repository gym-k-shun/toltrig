cmake_minimum_required(VERSION 3.16)
project(toltrig VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
add_library(toltrig INTERFACE)
target_include_directories(toltrig INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(toltrig INTERFACE cxx_std_17)
function(toltrig_exe name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE toltrig)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /permissive- /O2)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -O3)
  endif()
endfunction()
toltrig_exe(toltrig_tests tests/test_cos.cpp)
toltrig_exe(toltrig_tan_tests tests/test_tan.cpp)
toltrig_exe(toltrig_sin_tests tests/test_sin.cpp)
toltrig_exe(toltrig_bench bench/bench_cos.cpp)
toltrig_exe(toltrig_diagnostic bench/diagnostic_cos.cpp)
toltrig_exe(toltrig_tan_bench bench/bench_tan.cpp)
toltrig_exe(toltrig_tan_diagnostic bench/diagnostic_tan.cpp)
toltrig_exe(toltrig_sin_bench bench/bench_sin.cpp)
toltrig_exe(toltrig_sin_diagnostic bench/diagnostic_sin.cpp)
toltrig_exe(toltrig_example examples/basic.cpp)
enable_testing()
add_test(NAME toltrig_tests COMMAND toltrig_tests)
add_test(NAME toltrig_tan_tests COMMAND toltrig_tan_tests)
add_test(NAME toltrig_sin_tests COMMAND toltrig_sin_tests)
install(TARGETS toltrig EXPORT toltrigTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT toltrigTargets NAMESPACE toltrig:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/toltrig)
